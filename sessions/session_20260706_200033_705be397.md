import datetime

today_str = datetime.date.today().strftime("%Y-%m-%d")
daily_snapshot_path = f"/workspace/saves/daily/{today_str}.md"
content_to_save = f"# Daily Snapshot - {today_str}\n\n## 保存ファイル一覧\n"

# my-ai-notes.md の存在確認と内容読み込み
try:
    my_notes_content = default_api.read_file(path="/workspace/my-ai-notes.md")
    if my_notes_content and "content" in my_notes_content:
        lines = my_notes_content["content"].split('\n')
        content_to_save += f"- my-ai-notes.md ({len(lines)}行)\n"
    else:
        content_to_save += "- my-ai-notes.md (ファイルなしまたは空)\n"
except FileNotFoundError:
    content_to_save += "- my-ai-notes.md (ファイルなし)\n"
except Exception as e:
    content_to_save += f"- my-ai-notes.md (読み込みエラー: {e})\n"

# yamada-ai-knowledge-2026.md の存在確認と内容読み込み
try:
    yamada_notes_content = default_api.read_file(path="/workspace/yamada-ai-knowledge-2026.md")
    if yamada_notes_content and "content" in yamada_notes_content:
        lines = yamada_notes_content["content"].split('\n')
        content_to_save += f"- yamada-ai-knowledge-2026.md ({len(lines)}行)\n"
    else:
        content_to_save += "- yamada-ai-knowledge-2026.md (ファイルなしまたは空)\n"
except FileNotFoundError:
    content_to_save += "- yamada-ai-knowledge-2026.md (ファイルなし)\n"
except Exception as e:
    content_to_save += f"- yamada-ai-knowledge-2026.md (読み込みエラー: {e})\n"

# session_searchで直近の学習内容を取得
try:
    search_result = default_api.session_search(query="覚えた OR 学び OR 保存", limit=5, sort="newest")
    learnings = []
    if search_result and "sessions" in search_result:
        for session in search_result["sessions"]:
            if "messages" in session:
                for msg in session["messages"]:
                    if msg["role"] == "user" or msg["role"] == "assistant":
                         # メッセージ内容から関連する部分を抽出（簡易的な処理）
                        if any(keyword in msg["content"] for keyword in ["覚えた", "学び", "保存"]):
                            learnings.append(f"- {msg['content'].replace('\n', ' ').strip()[:100]} (セッション: {session.get('title', 'N/A')})")
        if learnings:
            content_to_save += "\n## 今日の学び（session_searchから）\n" + "\n".join(learnings)
        else:
            content_to_save += "\n## 今日の学び（session_searchから）\n該当する学習内容は見つかりませんでした。\n"
    else:
        content_to_save += "\n## 今日の学び（session_searchから）\nセッション履歴の検索に失敗しました。\n"
except Exception as e:
    content_to_save += f"\n## 今日の学び（session_searchから）\nセッション検索中にエラーが発生しました: {e}\n"

# ファイルに保存
try:
    # まず、念のためファイルの内容を読み込んで、もしあればその内容も結合する
    existing_content = ""
    try:
        existing_content_data = default_api.read_file(path=daily_snapshot_path)
        if existing_content_data and "content" in existing_content_data:
            existing_content = existing_content_data["content"] + "\n\n"
    except FileNotFoundError:
        pass # ファイルがなければ何もしない

    # 保存する内容を結合
    final_content = existing_content + content_to_save
    
    write_result = default_api.write_file(path=daily_snapshot_path, content=final_content)
    if write_result and "path" in write_result:
        print(f"Successfully saved snapshot to: {write_result['path']}")
        snapshot_save_status = "成功"
    else:
        print(f"Failed to save snapshot to: {daily_snapshot_path}")
        snapshot_save_status = f"失敗 (詳細: {write_result})"

except Exception as e:
    print(f"Error writing to {daily_snapshot_path}: {e}")
    snapshot_save_status = f"失敗 (エラー: {e})"

print(f"Snapshot save status: {snapshot_save_status}")

# 保存したファイルの内容を確認（デバッグ用）
try:
    final_snapshot_content = default_api.read_file(path=daily_snapshot_path)
    print("\n--- Saved Snapshot Content ---")
    print(final_snapshot_content["content"] if final_snapshot_content and "content" in final_snapshot_content else "ファイルの内容を読み込めませんでした。")
    print("----------------------------")
except Exception as e:
    print(f"Could not read back saved snapshot for verification: {e}")
